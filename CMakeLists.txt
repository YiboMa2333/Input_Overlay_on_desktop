cmake_minimum_required(VERSION 3.25)
project(InputOverlayDesktop VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(CTest)
option(OVERLAY_BUILD_APP "Build Windows desktop executable" ON)
option(OVERLAY_ENABLE_DESKTOP_TESTS "Run interactive Windows tests (moves pointer, opens backdrop)" OFF)
add_library(overlay_state src/input/input_state.cpp)
target_include_directories(overlay_state PUBLIC src)
target_compile_features(overlay_state PUBLIC cxx_std_20)
if(BUILD_TESTING)
  add_executable(input_state_tests tests/input_state_tests.cpp)
  target_link_libraries(input_state_tests PRIVATE overlay_state)
  add_test(NAME input_state_transitions COMMAND input_state_tests)
  add_executable(geometry_tests tests/geometry_tests.cpp)
  target_link_libraries(geometry_tests PRIVATE overlay_state)
  add_test(NAME placement_geometry COMMAND geometry_tests)
endif()
if(OVERLAY_BUILD_APP)
  if(NOT WIN32)
    message(FATAL_ERROR "Input Overlay Desktop requires Windows and the Windows SDK")
  endif()
  enable_language(RC)
  add_executable(InputOverlayDesktop WIN32
    src/main.cpp src/input/raw_input.cpp
    src/render/renderer.cpp src/settings/settings.cpp src/settings/dialog.cpp src/desktop/window.cpp resources/app.rc)
  target_include_directories(InputOverlayDesktop PRIVATE resources)
  target_compile_features(InputOverlayDesktop PRIVATE cxx_std_20)
  target_compile_definitions(InputOverlayDesktop PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN
    WINVER=0x0A00 _WIN32_WINNT=0x0A00)
  target_include_directories(InputOverlayDesktop PRIVATE src)
  target_link_libraries(InputOverlayDesktop PRIVATE overlay_state user32 gdi32 d2d1 dwrite wtsapi32 shell32 comctl32 comdlg32 advapi32)
  if(MSVC)
    target_compile_options(InputOverlayDesktop PRIVATE /W4 /permissive- /utf-8)
    target_link_options(InputOverlayDesktop PRIVATE /MANIFEST:NO)
  endif()
  if(MINGW)
    target_link_options(InputOverlayDesktop PRIVATE -municode -static "-Wl,-Map,$<TARGET_FILE_DIR:InputOverlayDesktop>/InputOverlayDesktop.map")
  endif()
  if(BUILD_TESTING)
    add_executable(settings_tests tests/settings_tests.cpp src/settings/settings.cpp)
    target_include_directories(settings_tests PRIVATE src)
    target_compile_features(settings_tests PRIVATE cxx_std_20)
    target_compile_definitions(settings_tests PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_link_libraries(settings_tests PRIVATE advapi32)
    add_test(NAME settings_persistence COMMAND settings_tests)
    find_package(Python3 COMPONENTS Interpreter)
    if(Python3_Interpreter_FOUND)
      add_test(NAME packaging_policy COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/package_tests.py $<TARGET_FILE:InputOverlayDesktop>)
    endif()
    add_executable(windows_integration_tests tests/windows_integration_tests.cpp)
    target_include_directories(windows_integration_tests PRIVATE src)
    target_compile_features(windows_integration_tests PRIVATE cxx_std_20)
    target_compile_definitions(windows_integration_tests PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN
      WINVER=0x0A00 _WIN32_WINNT=0x0A00)
    target_link_libraries(windows_integration_tests PRIVATE user32 gdi32 shell32 advapi32 ole32 oleaut32 uiautomationcore uuid)
    if(MINGW)
      target_link_options(windows_integration_tests PRIVATE -municode -static)
    endif()
    if(OVERLAY_ENABLE_DESKTOP_TESTS)
      add_test(NAME windows_desktop_integration COMMAND windows_integration_tests $<TARGET_FILE:InputOverlayDesktop>)
      set_tests_properties(windows_desktop_integration PROPERTIES RUN_SERIAL TRUE TIMEOUT 90)
    endif()
  endif()
  install(TARGETS InputOverlayDesktop RUNTIME DESTINATION .)
  install(FILES LICENSE README.md NOTICE.md docs/INSTALL.md DESTINATION .)
  install(DIRECTORY LICENSES docs DESTINATION .)
endif()
